       01  WS-SERVICE-TABLE.
           05 ST-COUNT            PIC 9(04) VALUE 0.
           05 ST-ENTRY OCCURS 200 TIMES INDEXED BY ST-IDX.

              PERFORM 1500-LOAD-SERVICES
           END-IF
           PERFORM 2000-PROCESS UNTIL END-OF-RESV
           PERFORM 3000-CLOSE-AND-REPORT
           STOP RUN.

       1000-OPEN-FILES.
           MOVE FUNCTION CURRENT-DATE (1:8) TO WS-TODAY
           OPEN INPUT RESV-FILE
           IF WS-FS1 NOT = '00'
              DISPLAY 'HRBSBIL: RESV-FILE OPEN ERROR FS=' WS-FS1
              MOVE 16 TO RETURN-CODE
              STOP RUN
           END-IF

           OPEN INPUT SRV-FILE
           IF WS-FS-SRV = '00'
              SET SRV-FILE-PRESENT TO TRUE
           ELSE
              IF WS-FS-SRV = '05'
                 DISPLAY 'HRBSBIL: OPTIONAL SERVICE FILE NOT PRESENT'
              ELSE
                 DISPLAY 'HRBSBIL: SERVICE FILE OPEN ERROR FS='
                         WS-FS-SRV
                 MOVE 16 TO RETURN-CODE
                 STOP RUN
              END-IF
           END-IF

           IF WS-FS2 NOT = '00'

              NOT AT END ADD 1 TO WS-READ-CNT
           END-READ.

           READ SRV-FILE
              AT END SET END-OF-SRV TO TRUE
           END-READ
           PERFORM UNTIL END-OF-SRV OR ST-COUNT >= 200
              ADD 1 TO ST-COUNT
              MOVE SRV-RES-ID TO ST-RES-ID (ST-COUNT)
              MOVE SRV-CODE   TO ST-CODE   (ST-COUNT)
              MOVE SRV-AMOUNT TO ST-AMOUNT (ST-COUNT)

                 AT END SET END-OF-SRV TO TRUE
              END-READ

           ELSE
              ADD 1 TO WS-SKIP-CNT
           END-IF

       2100-BUILD-INVOICE.
      *---- CALCULATE NIGHTS (MINIMUM 1) -----------------------------*
           COMPUTE WS-NIGHTS =
               FUNCTION INTEGER-OF-DATE (RV-CHKOUT)
             - FUNCTION INTEGER-OF-DATE (RV-CHKIN)
           IF WS-NIGHTS < 1
              MOVE 1 TO WS-NIGHTS
           END-IF

      *---- BASE ROOM CHARGES ----------------------------------------*
           COMPUTE WS-ROOM = WS-NIGHTS * RV-RATE
           COMPUTE WS-BASE-SVC ROUNDED =
               WS-ROOM * PARM-BASE-SVC-RATE

      *---- EXTRA SERVICE CHARGES FROM TABLE -------------------------*
           MOVE 0 TO WS-EXTRA-SVC
           IF SRV-FILE-PRESENT AND ST-COUNT > 0
              PERFORM VARYING ST-IDX FROM 1 BY 1
                      UNTIL ST-IDX > ST-COUNT
                 IF ST-RES-ID (ST-IDX) = RV-ID
                    ADD ST-AMOUNT (ST-IDX) TO WS-EXTRA-SVC
                 END-IF
              END-PERFORM
           END-IF

           COMPUTE WS-SVC = WS-BASE-SVC + WS-EXTRA-SVC

      *---- TIERED GST CALCULATION -----------------------------------*
           IF RV-RATE > PARM-GST-SLAB-LIMIT
              MOVE PARM-GST-HIGH-RATE TO WS-TAX-RATE
           ELSE
              MOVE PARM-GST-LOW-RATE  TO WS-TAX-RATE
           END-IF

           COMPUTE WS-TAX ROUNDED =
               (WS-ROOM + WS-SVC) * WS-TAX-RATE
           COMPUTE WS-TOT = WS-ROOM + WS-SVC + WS-TAX

      *---- FORMAT 100-BYTE INVOICE RECORD ---------------------------*
           ADD 1 TO WS-SEQ
           ADD 1 TO WS-GUEST-SEQ

           MOVE 'INV'              TO INV-TYPE
           MOVE RV-HOTEL           TO INV-HOTEL
           MOVE WS-TODAY           TO INV-DATE
           STRING 'G' WS-GUEST-SEQ DELIMITED BY SIZE INTO INV-GUEST
           STRING 'IN' WS-SEQ      DELIMITED BY SIZE INTO INV-ID
           MOVE RV-ID              TO INV-RES
           MOVE RV-NAME            TO INV-NAME
           MOVE WS-NIGHTS          TO INV-NIGHTS
           MOVE WS-ROOM            TO INV-ROOM
           MOVE WS-SVC             TO INV-SVC
           MOVE WS-TAX             TO INV-TAX
           MOVE WS-TOT             TO INV-TOT
           MOVE 'P'                TO INV-STATUS
           MOVE SPACES             TO FILLER OF INV-REC

           WRITE INV-REC
           ADD 1 TO WS-BILL-CNT
           ADD WS-TOT TO WS-GRAND-TOTAL.

       3000-CLOSE-AND-REPORT.
           CLOSE RESV-FILE INV-FILE
           IF SRV-FILE-PRESENT
              CLOSE SRV-FILE
           END-IF
           MOVE WS-GRAND-TOTAL TO WS-GRAND-EDIT
           DISPLAY '-------------------------------------------'
           DISPLAY 'HRBSBIL - BILLING RUN SUMMARY'
           DISPLAY 'RESERVATIONS READ : ' WS-READ-CNT
           DISPLAY 'INVOICES WRITTEN  : ' WS-BILL-CNT
           DISPLAY 'SKIPPED (NOT CONF): ' WS-SKIP-CNT
           DISPLAY 'EXTRA SRV LOADED  : ' WS-SRV-CNT
           DISPLAY 'TOTAL BILLED      : ' WS-GRAND-EDIT
           DISPLAY '-------------------------------------------'.

       IDENTIFICATION DIVISION.
       PROGRAM-ID. HRBSVAL.
      *****************************************************************
      * HOTEL RESERVATION AND BILLING SYSTEM (HRBS)
      * VALIDATION MODULE - VALIDATES SORTED RESERVATIONS (JOB01 OUTPUT)
      *
      * INPUT  (RESVIN) : RESERVATIONS, 80 BYTES (SORTED BY KEY)
      * OUTPUT (VALOUT) : VALID RESERVATIONS, 80 BYTES
      * OUTPUT (REJOUT) : REJECTED RESERVATIONS, 120 BYTES
      *                   (80-BYTE RECORD + 2-BYTE CODE + 38-BYTE MSG)
      *
      * VALIDATION RULES:
      * 1. RECORD TYPE = 'RSV'
      * 2. RESERVATION ID FORMAT = 'RS' + 6 NUMERIC DIGITS
      * 3. BOOKING DATE = VALID YYMMDD
      * 4. CHECK-IN DATE = VALID YYYYMMDD
      * 5. CHECK-OUT DATE = VALID YYYYMMDD
      * 6. CHECK-OUT DATE STRICTLY AFTER CHECK-IN DATE
      * 7. ROOM TYPE IN ('STD ', 'DLX ', 'STE ')
      * 8. GUEST COUNT BETWEEN 1 AND 4
      * 9. ROOM RATE CONSISTENT WITH TYPE (STD=3500, DLX=5200, STE=8800)
      * 10. STATUS IN ('C', 'X', 'O')
      *
      * RETURN CODES:
      *   00 = ALL RECORDS VALID (0 REJECTS)
      *   04 = ONE OR MORE RECORDS REJECTED

      *****************************************************************

       INPUT-OUTPUT SECTION.

           SELECT RESV-FILE ASSIGN TO RESVIN
               ORGANIZATION IS SEQUENTIAL
               FILE STATUS IS WS-FS-IN.
           SELECT VAL-FILE  ASSIGN TO VALOUT
               ORGANIZATION IS SEQUENTIAL
               FILE STATUS IS WS-FS-VAL.
           SELECT REJ-FILE  ASSIGN TO REJOUT
               ORGANIZATION IS SEQUENTIAL
               FILE STATUS IS WS-FS-REJ.

           05 RV-HOTEL            PIC X(03).
           05 RV-BKDATE           PIC X(06).
           05 RV-ID.
              10 RV-ID-PREFIX     PIC X(02).
              10 RV-ID-NUM        PIC 9(06).
           05 RV-NAME             PIC X(20).
           05 RV-ROOM             PIC X(04).
           05 RV-RTYPE            PIC X(04).
           05 RV-CHKIN            PIC 9(08).
           05 RV-CHKOUT           PIC 9(08).
           05 RV-GUESTS           PIC 9(02).
           05 RV-RATE             PIC 9(06).
           05 RV-STATUS           PIC X(01).
           05 FILLER              PIC X(07).

       FD  VAL-FILE
           RECORDING MODE IS F
           BLOCK CONTAINS 0 RECORDS.
       01  VAL-REC                PIC X(80).

       FD  REJ-FILE
           RECORDING MODE IS F
           BLOCK CONTAINS 0 RECORDS.
       01  REJ-REC.

       01  WS-FILE-STATUS.

              88 END-OF-INPUT     VALUE 'Y'.

           IF REC-IS-VALID AND RV-TYPE NOT = 'RSV'
              SET REC-IS-INVALID TO TRUE
              MOVE '01' TO WS-REJ-CODE
              MOVE 'INVALID RECORD TYPE (EXPECTED RSV)    '
                TO WS-REJ-MSG
           END-IF

      *---- RULE 2: RESV ID MUST BE 'RS' + 6 NUMERIC DIGITS ----------*
           IF REC-IS-VALID
              IF RV-ID-PREFIX NOT = 'RS'
                 OR RV-ID-NUM NOT NUMERIC
                 SET REC-IS-INVALID TO TRUE
                 MOVE '02' TO WS-REJ-CODE
                 MOVE 'INVALID RES-ID FORMAT (RS+6 DIGITS)  '
                   TO WS-REJ-MSG
              END-IF
           END-IF

      *---- RULE 3: CHECK-IN DATE VALIDITY ---------------------------*
           IF REC-IS-VALID
              MOVE RV-CHKIN (1:4) TO DW-YYYY
              MOVE RV-CHKIN (5:2) TO DW-MM
              MOVE RV-CHKIN (7:2) TO DW-DD
              PERFORM 2500-VALIDATE-CALENDAR-DATE
              IF REC-IS-INVALID
                 MOVE '04' TO WS-REJ-CODE
                 MOVE 'INVALID CHECK-IN DATE (YYYYMMDD)     '
                   TO WS-REJ-MSG
              END-IF
           END-IF

      *---- RULE 4: CHECK-OUT DATE VALIDITY --------------------------*
           IF REC-IS-VALID
              MOVE RV-CHKOUT (1:4) TO DW-YYYY
              MOVE RV-CHKOUT (5:2) TO DW-MM
              MOVE RV-CHKOUT (7:2) TO DW-DD
              PERFORM 2500-VALIDATE-CALENDAR-DATE
              IF REC-IS-INVALID
                 MOVE '05' TO WS-REJ-CODE
                 MOVE 'INVALID CHECK-OUT DATE (YYYYMMDD)    '
                   TO WS-REJ-MSG
              END-IF
           END-IF

      *---- RULE 5: CHECK-OUT STRICTLY AFTER CHECK-IN ----------------*
           IF REC-IS-VALID
              IF RV-CHKOUT <= RV-CHKIN
                 SET REC-IS-INVALID TO TRUE
                 MOVE '06' TO WS-REJ-CODE
                 MOVE 'CHECK-OUT DATE NOT AFTER CHECK-IN    '
                   TO WS-REJ-MSG
              END-IF
           END-IF

           IF REC-IS-VALID
              IF RV-RTYPE NOT = 'STD ' AND RV-RTYPE NOT = 'DLX '
                 AND RV-RTYPE NOT = 'STE '
                 SET REC-IS-INVALID TO TRUE
                 MOVE '07' TO WS-REJ-CODE
                 MOVE 'INVALID ROOM TYPE (NOT STD/DLX/STE)  '
                   TO WS-REJ-MSG
              END-IF
           END-IF

           IF REC-IS-VALID
              IF RV-GUESTS < 1 OR RV-GUESTS > 4
                 SET REC-IS-INVALID TO TRUE
                 MOVE '08' TO WS-REJ-CODE
                 MOVE 'GUEST COUNT OUT OF RANGE (1 - 4)     '
                   TO WS-REJ-MSG
              END-IF
           END-IF

      *---- RULE 8: RATE CONSISTENCY WITH ROOM TYPE ------------------*
           IF REC-IS-VALID
              EVALUATE RV-RTYPE
                 WHEN 'STD '
                    IF RV-RATE NOT = 3500
                       SET REC-IS-INVALID TO TRUE
                    END-IF
                 WHEN 'DLX '
                    IF RV-RATE NOT = 5200
                       SET REC-IS-INVALID TO TRUE
                    END-IF
                 WHEN 'STE '
                    IF RV-RATE NOT = 8800
                       SET REC-IS-INVALID TO TRUE
                    END-IF
                 WHEN OTHER
                    SET REC-IS-INVALID TO TRUE
              END-EVALUATE
              IF REC-IS-INVALID
                 MOVE '09' TO WS-REJ-CODE
                 MOVE 'RATE INCONSISTENT WITH ROOM TYPE     '
                   TO WS-REJ-MSG
              END-IF
           END-IF

      *---- RULE 9: STATUS MUST BE C, X, OR O ------------------------*
           IF REC-IS-VALID
              IF RV-STATUS NOT = 'C' AND RV-STATUS NOT = 'X'
                 AND RV-STATUS NOT = 'O'
                 SET REC-IS-INVALID TO TRUE
                 MOVE '10' TO WS-REJ-CODE
                 MOVE 'INVALID STATUS (NOT C, X, OR O)      '
                   TO WS-REJ-MSG
              END-IF
           END-IF

      *---- DISPOSITION ----------------------------------------------*
           IF REC-IS-VALID
              MOVE RESV-REC TO VAL-REC
              WRITE VAL-REC
              ADD 1 TO WS-VAL-CNT
           ELSE
              MOVE RESV-REC   TO RJ-ORIG-REC
              MOVE WS-REJ-CODE TO RJ-CODE
              MOVE WS-REJ-MSG  TO RJ-MSG
              WRITE REJ-REC
              ADD 1 TO WS-REJ-CNT
           END-IF

           PERFORM 1100-READ-INPUT.

       2500-VALIDATE-CALENDAR-DATE.
           IF DW-YYYY < 2020 OR DW-YYYY > 2035
              SET REC-IS-INVALID TO TRUE
           END-IF
           IF DW-MM < 1 OR DW-MM > 12
              SET REC-IS-INVALID TO TRUE
           END-IF
           IF REC-IS-VALID
              EVALUATE DW-MM
                 WHEN 1  MOVE 31 TO DW-MAX-DAYS
                 WHEN 3  MOVE 31 TO DW-MAX-DAYS
                 WHEN 5  MOVE 31 TO DW-MAX-DAYS
                 WHEN 7  MOVE 31 TO DW-MAX-DAYS
                 WHEN 8  MOVE 31 TO DW-MAX-DAYS
                 WHEN 10 MOVE 31 TO DW-MAX-DAYS
                 WHEN 12 MOVE 31 TO DW-MAX-DAYS
                 WHEN 4  MOVE 30 TO DW-MAX-DAYS
                 WHEN 6  MOVE 30 TO DW-MAX-DAYS
                 WHEN 9  MOVE 30 TO DW-MAX-DAYS
                 WHEN 11 MOVE 30 TO DW-MAX-DAYS
                 WHEN 2
                    DIVIDE DW-YYYY BY 4 GIVING DW-LEAP-REM
                    REMAINDER DW-LEAP-REM
                    IF DW-LEAP-REM = 0
                       MOVE 29 TO DW-MAX-DAYS
                    ELSE
                       MOVE 28 TO DW-MAX-DAYS
                    END-IF
              END-EVALUATE
              IF DW-DD < 1 OR DW-DD > DW-MAX-DAYS
                 SET REC-IS-INVALID TO TRUE
              END-IF
           END-IF.

           DISPLAY 'VALID (WRITTEN VAL)  : ' WS-VAL-CNT
           DISPLAY 'REJECTS (WRITTEN REJ): ' WS-REJ-CNT
           DISPLAY '-------------------------------------------'.

              AT END SET END-OF-TRN TO TRUE
              NOT AT END ADD 1 TO WS-TRN-READ
           END-READ.

       2000-PROCESS-TRANS.
           EVALUATE TR-ACTION
              WHEN 'A'
                 PERFORM 2100-HANDLE-ADD
              WHEN 'C'
                 PERFORM 2200-HANDLE-CANCEL
              WHEN 'O'
                 PERFORM 2300-HANDLE-CHECKOUT
              WHEN 'U'
                 PERFORM 2400-HANDLE-UPDATE
              WHEN OTHER
                 PERFORM 2900-HANDLE-INVALID-ACTION
           END-EVALUATE
           PERFORM 1100-READ-TRN.

       2100-HANDLE-ADD.
      *---- CHECK IF KEY ALREADY EXISTS IN KSDS ----------------------*
           MOVE TR-ID TO KS-ID
           READ MAST-FILE
           IF WS-FS-KSDS = '00'
              MOVE '22' TO WS-REJ-CODE
              MOVE 'DUPLICATE KEY - RESERVATION EXISTS    '
                TO WS-REJ-MSG
              PERFORM 2800-REJECT-TRANS
           ELSE
      *---- CHECK ROOM OVERLAP ---------------------------------------*
              PERFORM 2500-CHECK-ROOM-AVAILABILITY
              IF HAS-OVERLAP
                 MOVE 'RO' TO WS-REJ-CODE
                 MOVE 'ROOM NOT AVAILABLE - OVERLAPPING DATES '
                   TO WS-REJ-MSG
                 PERFORM 2800-REJECT-TRANS
              ELSE
                 PERFORM 2150-EXECUTE-ADD
              END-IF
           END-IF.

       2150-EXECUTE-ADD.
           MOVE SPACES         TO MAST-REC
           MOVE 'RSV'          TO KS-TYPE
           MOVE 'PUN'          TO KS-HOTEL
           MOVE WS-TODAY (3:6) TO KS-BKDATE
           MOVE TR-ID          TO KS-ID
           MOVE TR-NAME        TO KS-NAME
           MOVE TR-ROOM        TO KS-ROOM
           MOVE TR-RTYPE       TO KS-RTYPE
           MOVE TR-CHKIN       TO KS-CHKIN
           MOVE TR-CHKOUT      TO KS-CHKOUT
           MOVE TR-GUESTS      TO KS-GUESTS
           MOVE TR-RATE        TO KS-RATE
           MOVE 'C'            TO KS-STATUS
           MOVE SPACES         TO KS-FILLER

           WRITE MAST-REC
           IF WS-FS-KSDS = '00'
              ADD 1 TO WS-ADD-CNT
              MOVE 'ADD   '                       TO WS-AUD-ACT
              MOVE 'OK  '                         TO WS-AUD-RES
              MOVE 'RESERVATION ADDED SUCCESSFULLY' TO WS-AUD-MSG
              PERFORM 2700-WRITE-AUDIT
              IF BT-COUNT < 100
                 ADD 1 TO BT-COUNT
                 MOVE TR-ROOM   TO BT-ROOM (BT-COUNT)
                 MOVE TR-CHKIN  TO BT-CHKIN (BT-COUNT)
                 MOVE TR-CHKOUT TO BT-CHKOUT (BT-COUNT)
                 MOVE 'C'       TO BT-STATUS (BT-COUNT)
              END-IF
           ELSE
              MOVE WS-FS-KSDS TO WS-REJ-CODE
              MOVE 'VSAM WRITE FAILED                     '
                TO WS-REJ-MSG
              PERFORM 2800-REJECT-TRANS
           END-IF.

       2200-HANDLE-CANCEL.
           MOVE TR-ID TO KS-ID
           READ MAST-FILE
           IF WS-FS-KSDS = '23'
              MOVE '23' TO WS-REJ-CODE
              MOVE 'RESERVATION NOT FOUND FOR CANCELLATION'
                TO WS-REJ-MSG
              PERFORM 2800-REJECT-TRANS
           ELSE
              IF WS-FS-KSDS = '00'
                 MOVE 'X' TO KS-STATUS
                 REWRITE MAST-REC
                 IF WS-FS-KSDS = '00'
                    ADD 1 TO WS-CAN-CNT
                    MOVE 'CANCEL'                 TO WS-AUD-ACT
                    MOVE 'OK  '                   TO WS-AUD-RES
                    MOVE 'RESERVATION CANCELLED   ' TO WS-AUD-MSG
                    PERFORM 2700-WRITE-AUDIT
                 ELSE
                    MOVE WS-FS-KSDS TO WS-REJ-CODE
                    MOVE 'VSAM REWRITE FAILED                   '
                      TO WS-REJ-MSG
                    PERFORM 2800-REJECT-TRANS
                 END-IF
              ELSE
                 MOVE WS-FS-KSDS TO WS-REJ-CODE
                 MOVE 'VSAM READ ERROR                       '
                   TO WS-REJ-MSG
                 PERFORM 2800-REJECT-TRANS
              END-IF
           END-IF.

       2300-HANDLE-CHECKOUT.
           MOVE TR-ID TO KS-ID
           READ MAST-FILE
           IF WS-FS-KSDS = '23'
              MOVE '23' TO WS-REJ-CODE
              MOVE 'RESERVATION NOT FOUND FOR CHECK-OUT   '
                TO WS-REJ-MSG
              PERFORM 2800-REJECT-TRANS
           ELSE
              IF WS-FS-KSDS = '00'
                 MOVE 'O' TO KS-STATUS
                 REWRITE MAST-REC
                 IF WS-FS-KSDS = '00'
                    ADD 1 TO WS-OUT-CNT
                    MOVE 'CHKOUT'                 TO WS-AUD-ACT
                    MOVE 'OK  '                   TO WS-AUD-RES
                    MOVE 'GUEST CHECKED OUT       ' TO WS-AUD-MSG
                    PERFORM 2700-WRITE-AUDIT
                 ELSE
                    MOVE WS-FS-KSDS TO WS-REJ-CODE
                    MOVE 'VSAM REWRITE FAILED                   '
                      TO WS-REJ-MSG
                    PERFORM 2800-REJECT-TRANS
                 END-IF
              ELSE
                 MOVE WS-FS-KSDS TO WS-REJ-CODE
                 MOVE 'VSAM READ ERROR                       '
                   TO WS-REJ-MSG
                 PERFORM 2800-REJECT-TRANS
              END-IF
           END-IF.

       2400-HANDLE-UPDATE.
           MOVE TR-ID TO KS-ID
           READ MAST-FILE
           IF WS-FS-KSDS = '23'
              MOVE '23' TO WS-REJ-CODE
              MOVE 'RESERVATION NOT FOUND FOR UPDATE      '
                TO WS-REJ-MSG
              PERFORM 2800-REJECT-TRANS
           ELSE
              IF WS-FS-KSDS = '00'
                 IF TR-RATE > 0
                    MOVE TR-RATE TO KS-RATE
                 END-IF
                 IF TR-CHKIN > 0
                    MOVE TR-CHKIN TO KS-CHKIN
                 END-IF
                 IF TR-CHKOUT > 0
                    MOVE TR-CHKOUT TO KS-CHKOUT
                 END-IF
                 REWRITE MAST-REC
                 IF WS-FS-KSDS = '00'
                    ADD 1 TO WS-UPD-CNT
                    MOVE 'UPDATE'                 TO WS-AUD-ACT
                    MOVE 'OK  '                   TO WS-AUD-RES
                    MOVE 'RESERVATION DETAILS UPDATED'
                      TO WS-AUD-MSG
                    PERFORM 2700-WRITE-AUDIT
                 ELSE
                    MOVE WS-FS-KSDS TO WS-REJ-CODE
                    MOVE 'VSAM REWRITE FAILED                   '
                      TO WS-REJ-MSG
                    PERFORM 2800-REJECT-TRANS
                 END-IF
              ELSE
                 MOVE WS-FS-KSDS TO WS-REJ-CODE
                 MOVE 'VSAM READ ERROR                       '
                   TO WS-REJ-MSG
                 PERFORM 2800-REJECT-TRANS
              END-IF
           END-IF.

       2500-CHECK-ROOM-AVAILABILITY.
           MOVE 'N' TO WS-OVERLAP-FOUND
           IF BT-COUNT > 0
              PERFORM VARYING BT-IDX FROM 1 BY 1
                      UNTIL BT-IDX > BT-COUNT OR HAS-OVERLAP
                 IF BT-ROOM (BT-IDX) = TR-ROOM
                    AND BT-STATUS (BT-IDX) NOT = 'X'
                    IF TR-CHKIN < BT-CHKOUT (BT-IDX)
                       AND TR-CHKOUT > BT-CHKIN (BT-IDX)
                       SET HAS-OVERLAP TO TRUE
                    END-IF
                 END-IF
              END-PERFORM
           END-IF.

       2700-WRITE-AUDIT.
           MOVE WS-TODAY   TO AU-DATE
           MOVE WS-AUD-ACT TO AU-ACTION
           MOVE TR-ID      TO AU-ID
           MOVE TR-ROOM    TO AU-ROOM
           MOVE WS-AUD-RES TO AU-RESULT
           MOVE WS-AUD-MSG TO AU-MESSAGE
           WRITE AUD-REC.

       2800-REJECT-TRANS.
           MOVE TRN-REC    TO RJ-ORIG-TRN
           MOVE WS-REJ-CODE TO RJ-CODE
           MOVE WS-REJ-MSG  TO RJ-REASON
           WRITE REJ-REC
           ADD 1 TO WS-REJ-CNT
           MOVE 'REJ   '   TO WS-AUD-ACT
           MOVE 'FAIL'     TO WS-AUD-RES
           MOVE WS-REJ-MSG TO WS-AUD-MSG
           PERFORM 2700-WRITE-AUDIT.

       2900-HANDLE-INVALID-ACTION.
           MOVE '99' TO WS-REJ-CODE
           MOVE 'INVALID TRANSACTION ACTION CODE       '
             TO WS-REJ-MSG
           PERFORM 2800-REJECT-TRANS.

       3000-CLOSE-AND-REPORT.
           CLOSE MAST-FILE TRN-FILE AUD-FILE REJ-FILE
           DISPLAY '-------------------------------------------'
           DISPLAY 'HRBSUPD - VSAM TRANSACTION UPDATE SUMMARY'
           DISPLAY 'TRANSACTIONS READ   : ' WS-TRN-READ
           DISPLAY 'ADDS PROCESSED      : ' WS-ADD-CNT
           DISPLAY 'CANCELS PROCESSED   : ' WS-CAN-CNT
           DISPLAY 'CHECK-OUTS PROCESSED: ' WS-OUT-CNT
           DISPLAY 'UPDATES PROCESSED   : ' WS-UPD-CNT
           DISPLAY 'REJECTED TRANS      : ' WS-REJ-CNT
           DISPLAY '-------------------------------------------'.

                  FUNCTION CURRENT-DATE (7:2)
                  DELIMITED BY SIZE INTO WS-TODAY
           OPEN INPUT  IN-FILE
           OPEN OUTPUT RPT-FILE
           IF WS-FS-IN NOT = '00' OR WS-FS-RPT NOT = '00'
              DISPLAY 'HRBSRPT: OPEN ERROR FS-IN=' WS-FS-IN
                      ' FS-RPT=' WS-FS-RPT
              MOVE 12 TO RETURN-CODE
              STOP RUN
           END-IF
           PERFORM 1100-READ-INVOICE.

       1100-READ-INVOICE.
           READ IN-FILE
              AT END SET END-OF-FILE TO TRUE
           END-READ.

       2000-PROCESS.
      *---- DETERMINE ROOM TYPE (FROM EXTENSION OR INFERRED) ----------*
           IF IR-RTYPE-EXT = 'STD ' OR IR-RTYPE-EXT = 'DLX '
              OR IR-RTYPE-EXT = 'STE '
              MOVE IR-RTYPE-EXT TO WS-CURRENT-RTYPE
           ELSE
              PERFORM 2050-INFER-ROOM-TYPE
           END-IF

      *---- CHECK FOR CONTROL BREAK ON ROOM TYPE ---------------------*
           IF FIRST-RECORD
              MOVE WS-CURRENT-RTYPE TO WS-PREV-RTYPE
              MOVE 'N' TO WS-FIRST-REC
           ELSE
              IF WS-CURRENT-RTYPE NOT = WS-PREV-RTYPE
                 PERFORM 2800-PRINT-SUBTOTAL
                 MOVE WS-CURRENT-RTYPE TO WS-PREV-RTYPE
              END-IF
           END-IF

           PERFORM 2100-PRINT-DETAIL
           PERFORM 2200-ACCUMULATE-TOTALS
           PERFORM 1100-READ-INVOICE.

       2050-INFER-ROOM-TYPE.
           IF IR-NIGHTS > 0
              COMPUTE WS-SUB-TOTALS = IR-ROOM / IR-NIGHTS
              IF SUB-COUNT <= 4000
                 MOVE 'STD ' TO WS-CURRENT-RTYPE
              ELSE
                 IF SUB-COUNT <= 6000
                    MOVE 'DLX ' TO WS-CURRENT-RTYPE
                 ELSE
                    MOVE 'STE ' TO WS-CURRENT-RTYPE
                 END-IF
              END-IF
           ELSE
              MOVE 'STD ' TO WS-CURRENT-RTYPE
           END-IF.

       2100-PRINT-DETAIL.
           IF WS-LINE-COUNT >= WS-MAX-LINES
              PERFORM 2500-PRINT-HEADINGS
           END-IF

           MOVE IR-ID            TO DT-INVID
           MOVE IR-GUEST         TO DT-GUESTID
           MOVE IR-NAME          TO DT-NAME
           MOVE IR-RES           TO DT-RESVID
           MOVE WS-CURRENT-RTYPE TO DT-RTYPE
           MOVE IR-NIGHTS        TO DT-NIGHTS
           MOVE IR-ROOM          TO DT-ROOM
           MOVE IR-SVC           TO DT-SVC
           MOVE IR-TAX           TO DT-TAX
           MOVE IR-TOT           TO DT-TOT
           IF IR-STATUS = 'P'
              MOVE 'PAID  ' TO DT-STATUS
           ELSE
              MOVE 'UNPAID' TO DT-STATUS
           END-IF

           WRITE RPT-REC FROM DETAIL-LINE
           ADD 1 TO WS-LINE-COUNT.

       2200-ACCUMULATE-TOTALS.
           ADD 1      TO SUB-COUNT
           ADD IR-ROOM TO SUB-ROOM
           ADD IR-SVC  TO SUB-SVC
           ADD IR-TAX  TO SUB-TAX
           ADD IR-TOT  TO SUB-TOT

           ADD 1      TO GRD-COUNT
           ADD IR-ROOM TO GRD-ROOM
           ADD IR-SVC  TO GRD-SVC
           ADD IR-TAX  TO GRD-TAX
           ADD IR-TOT  TO GRD-TOT.

           MOVE WS-PAGE-NUM TO H1-PAGE

              MOVE SPACES TO RPT-REC
              WRITE RPT-REC
           END-IF
           WRITE RPT-REC FROM HDG-LINE-1
           WRITE RPT-REC FROM HDG-LINE-2
           WRITE RPT-REC FROM DIVIDER-DOUBLE
           WRITE RPT-REC FROM COL-HEADINGS
           WRITE RPT-REC FROM DIVIDER-SINGLE
           MOVE 6 TO WS-LINE-COUNT.

       2800-PRINT-SUBTOTAL.
           IF SUB-COUNT > 0
              IF WS-LINE-COUNT + 4 >= WS-MAX-LINES
                 PERFORM 2500-PRINT-HEADINGS
              END-IF
              WRITE RPT-REC FROM DIVIDER-SINGLE
              MOVE WS-PREV-RTYPE TO ST-RTYPE
              MOVE SUB-COUNT     TO ST-COUNT
              MOVE SUB-ROOM      TO ST-ROOM
              MOVE SUB-SVC       TO ST-SVC
              MOVE SUB-TAX       TO ST-TAX
              MOVE SUB-TOT       TO ST-TOT
              WRITE RPT-REC FROM SUBTOTAL-LINE
              WRITE RPT-REC FROM DIVIDER-SINGLE
              MOVE SPACES TO RPT-REC
              WRITE RPT-REC
              ADD 4 TO WS-LINE-COUNT

              INITIALIZE WS-SUB-TOTALS
           END-IF.

       2900-PRINT-GRANDTOTAL.
           IF WS-LINE-COUNT + 5 >= WS-MAX-LINES
              PERFORM 2500-PRINT-HEADINGS
           END-IF
           WRITE RPT-REC FROM DIVIDER-DOUBLE
           MOVE GRD-COUNT TO GT-COUNT
           MOVE GRD-ROOM  TO GT-ROOM
           MOVE GRD-SVC   TO GT-SVC
           MOVE GRD-TAX   TO GT-TAX
           MOVE GRD-TOT   TO GT-TOT
           WRITE RPT-REC FROM GRANDTOTAL-LINE
           WRITE RPT-REC FROM DIVIDER-DOUBLE
           MOVE '**** END OF REPORT ****' TO RPT-REC
           WRITE RPT-REC.

       3000-CLOSE-FILES.
           CLOSE IN-FILE RPT-FILE
           DISPLAY '-------------------------------------------'
           DISPLAY 'HRBSRPT - BILLING REPORT SUMMARY'
           DISPLAY 'TOTAL INVOICES PRINTED: ' GRD-COUNT
           DISPLAY 'PAGES GENERATED       : ' WS-PAGE-NUM
           DISPLAY '-------------------------------------------'.

       IDENTIFICATION DIVISION.
       PROGRAM-ID. HRBSDB2.
      *****************************************************************
      * HOTEL RESERVATION AND BILLING SYSTEM (HRBS)
      * DB2 BATCH LOADER AND REPORTING MODULE (EMBEDDED SQL)
      *
      * INPUT  (RESVIN) : VALIDATED RESERVATION FILE (LRECL 80)

      *
      * FUNCTIONALITY:
      * 1. INSERTS RESERVATIONS INTO HRBS.RESERVATION TABLE.
      * 2. INSERTS INVOICES INTO HRBS.BILL TABLE.
      * 3. HANDLES DUPLICATE KEYS (SQLCODE -803) GRACEFULLY.
      * 4. ISSUES PERIODIC COMMIT (EVERY 25 ROWS) AND FINAL COMMIT.
      * 5. RUNS A CURSOR-BASED REPORT TO DISPLAY UNPAID BILLS.
      *
      * RETURN CODES:
      *   00 = NORMAL COMPLETION
      *   12 = FATAL DB2 SQL ERROR (ROLLBACK ISSUED)
      *****************************************************************
       ENVIRONMENT DIVISION.
       INPUT-OUTPUT SECTION.
       FILE-CONTROL.
           SELECT RESV-FILE ASSIGN TO RESVIN
               ORGANIZATION IS SEQUENTIAL
               FILE STATUS IS WS-FS-RESV.
           SELECT INV-FILE  ASSIGN TO INVIN
               ORGANIZATION IS SEQUENTIAL
               FILE STATUS IS WS-FS-INV.

       DATA DIVISION.
       FILE SECTION.
       FD  RESV-FILE
           RECORDING MODE IS F
           BLOCK CONTAINS 0 RECORDS.
       01  RESV-REC.
           05 RV-TYPE             PIC X(03).
           05 RV-HOTEL            PIC X(03).
           05 RV-BKDATE           PIC X(06).
           05 RV-ID               PIC X(08).
           05 RV-NAME             PIC X(20).

/* REXX */
/*********************************************************************/
/* HOTEL RESERVATION AND BILLING SYSTEM (HRBS) - ASSIGNMENT 3         */
/* READ INVOICE FILE (DD INDD, 100-BYTE RECORDS), REMOVE DUPLICATES,  */
/* SORT ON PRIMARY KEY (COLS 25-32) ASCENDING, WRITE TO DD OUTDD.     */
/* RUN IN BATCH (IKJEFT01) OR IN TSO AFTER ALLOCATING INDD / OUTDD.   */
/*********************************************************************/
keyStart = 25
keyLen   = 8

address TSO
"EXECIO * DISKR INDD (STEM in. FINIS"
read_rc = rc
if read_rc <> 0 & read_rc <> 2 then do
   say 'HRBSDEDP: READ ERROR, EXECIO RC='read_rc
   exit 8
end
total = in.0
say 'HRBSDEDP: RECORDS READ            =' total

/* ---- STEP A: REMOVE DUPLICATES (FIRST OCCURRENCE OF A KEY WINS) --- */
seen. = 0
n = 0
do i = 1 to total
   k = 'K' || substr(in.i, keyStart, keyLen)
   if seen.k = 0 then do
      seen.k = 1
      n = n + 1
      rec.n = in.i
      key.n = substr(in.i, keyStart, keyLen)
   end
end
dups = total - n
say 'HRBSDEDP: DUPLICATES REMOVED      =' dups
say 'HRBSDEDP: UNIQUE RECORDS          =' n

/* ---- STEP B: SORT ON KEY (SHELL SORT, STRICT STRING COMPARE) ------ */
gap = n % 2
do while gap > 0
   do i = gap + 1 to n
      tmpRec = rec.i
      tmpKey = key.i
      j = i
      do while j > gap
         p = j - gap
         if key.p >> tmpKey then do
            rec.j = rec.p
            key.j = key.p
            j = p
         end
         else leave
      end
      rec.j = tmpRec
      key.j = tmpKey
   end
   gap = gap % 2
end

/* ---- STEP C: WRITE RESULT ---------------------------------------- */
rec.0 = n
"EXECIO" n "DISKW OUTDD (STEM rec. FINIS"
if rc <> 0 then do
   say 'HRBSDEDP: WRITE ERROR, EXECIO RC='rc
   exit 12
end
say 'HRBSDEDP: RECORDS WRITTEN         =' n
say 'HRBSDEDP: COMPLETED SUCCESSFULLY'
exit 0

/* REXX */
/*********************************************************************/
/* HOTEL RESERVATION AND BILLING SYSTEM (HRBS)                       */
/* TSO COMMAND-LINE UTILITY (INVOKED FROM ISPF 6 OR TSO READY)       */
/*                                                                   */
/* SYNTAX:                                                           */
/*   TSO %HRBSUTIL COUNT  <dsname>                                   */
/*   TSO %HRBSUTIL CHKDUP <dsname> <start_col> <length>              */
/*   TSO %HRBSUTIL FIND   <dsname> <reservation_id>                  */
/*                                                                   */
/* EXAMPLES:                                                         */
/*   %HRBSUTIL COUNT  HRBS.RESERV.INPUT                              */
/*   %HRBSUTIL CHKDUP HRBS.RESERV.INPUT 13 8                         */
/*   %HRBSUTIL FIND   HRBS.RESERV.INPUT RS006881                     */
/*                                                                   */
/* RETURN CODES:                                                     */
/*   0 = SUCCESS                                                     */
/*   4 = WARNING (DUPLICATE FOUND / RECORD NOT FOUND)                */
/*   8 = INVALID SYNTAX / FILE ERROR                                 */
/*********************************************************************/
parse upper arg cmd dsn arg3 arg4 rest

if cmd = '' | cmd = 'HELP' | cmd = '?' then do
   call print_usage
   exit 0
end

if dsn = '' then do
   say 'HRBSUTIL: ERROR - DATASET NAME IS REQUIRED.'
   call print_usage
   exit 8
end

/* Ensure dataset name has quotes if not provided */
if left(dsn, 1) <> "'" then
   full_dsn = "'" || dsn || "'"
else
   full_dsn = dsn

select
   when cmd = 'COUNT'  then call do_count
   when cmd = 'CHKDUP' then call do_chkdup
   when cmd = 'FIND'   then call do_find
   otherwise do
      say 'HRBSUTIL: UNKNOWN COMMAND "'cmd'".'
      call print_usage
      exit 8
   end
end
exit 0

/* ----------------------------------------------------------------- */
/* DO_COUNT: COUNTS RECORDS IN A DATASET                             */
/* ----------------------------------------------------------------- */
do_count:
   call alloc_file
   address TSO "EXECIO * DISKR UTLDD (STEM in. FINIS"
   r_rc = rc
   if r_rc <> 0 & r_rc <> 2 then do
      say 'HRBSUTIL: READ ERROR ON ' full_dsn ', EXECIO RC=' r_rc
      call free_file
      exit 8
   end
   call free_file
   say '=================================================='
   say 'HRBSUTIL: DATASET RECORD COUNT SUMMARY'
   say 'DATASET  : ' full_dsn
   say 'RECORDS  : ' in.0
   say '=================================================='
   return

/* ----------------------------------------------------------------- */
/* DO_CHKDUP: CHECKS FOR DUPLICATES ON KEY RANGE                     */
/* ----------------------------------------------------------------- */
do_chkdup:
   if arg3 = '' | arg4 = '' then do
      say 'HRBSUTIL: ERROR - CHKDUP REQUIRES START_COL AND LENGTH.'
      say 'SYNTAX: %HRBSUTIL CHKDUP <dsn> <start_col> <length>'
      exit 8
   end
   k_start = arg3
   k_len   = arg4
   if \datatype(k_start, 'W') | \datatype(k_len, 'W') then do
      say 'HRBSUTIL: START_COL AND LENGTH MUST BE POSITIVE INTEGERS.'
      exit 8
   end

   call alloc_file
   address TSO "EXECIO * DISKR UTLDD (STEM in. FINIS"
   r_rc = rc
   if r_rc <> 0 & r_rc <> 2 then do
      say 'HRBSUTIL: READ ERROR ON ' full_dsn ', EXECIO RC=' r_rc
      call free_file
      exit 8
   end
   call free_file

   seen. = 0
   dup_cnt = 0
   say '=================================================='
   say 'HRBSUTIL: KEY DUPLICATE AUDIT REPORT'
   say 'DATASET  : ' full_dsn
   say 'KEY RANGE: COL ' k_start ' FOR ' k_len ' BYTES'
   say 'RECORDS  : ' in.0
   say '--------------------------------------------------'

   do i = 1 to in.0
      k = 'K' || substr(in.i, k_start, k_len)
      if seen.k > 0 then do
         dup_cnt = dup_cnt + 1
         say 'DUPLICATE #' dup_cnt ': KEY="'substr(in.i, k_start, k_len)'"' ,
             'AT RECORD' i ' (FIRST SEEN AT REC 'seen.k')'
      end
      else
         seen.k = i
   end

   say '--------------------------------------------------'
   if dup_cnt = 0 then do
      say 'RESULT: VERIFIED - ZERO DUPLICATES FOUND.'
      say '=================================================='
      return
   end
   else do
      say 'RESULT: AUDIT FAILED - ' dup_cnt ' DUPLICATES DETECTED.'
      say '=================================================='
      exit 4
   end

/* ----------------------------------------------------------------- */
/* DO_FIND: FINDS AND FORMATS A RESERVATION BY ID                    */
/* ----------------------------------------------------------------- */
do_find:
   if arg3 = '' then do
      say 'HRBSUTIL: ERROR - RESERVATION ID REQUIRED.'
      say 'SYNTAX: %HRBSUTIL FIND <dsn> <resid>'
      exit 8
   end
   find_id = arg3

   call alloc_file
   address TSO "EXECIO * DISKR UTLDD (STEM in. FINIS"
   r_rc = rc
   if r_rc <> 0 & r_rc <> 2 then do
      say 'HRBSUTIL: READ ERROR ON ' full_dsn ', EXECIO RC=' r_rc
      call free_file
      exit 8
   end
   call free_file

   found = 0
   do i = 1 to in.0
      /* Key is at cols 13-20 */
      if length(in.i) >= 20 then
         cur_id = substr(in.i, 13, 8)
      else
         cur_id = ''

      if cur_id = find_id then do
         found = 1
         r = in.i
         say '=================================================='
         say 'HRBSUTIL: RESERVATION RECORD LOCATED (REC #' i ')'
         say '=================================================='
         say 'RECORD TYPE : ' substr(r, 1, 3)
         say 'HOTEL CODE  : ' substr(r, 4, 3)
         say 'BOOKING DATE: 20' || substr(r, 7, 2) || '-' || ,
                              substr(r, 9, 2) || '-' || substr(r, 11, 2)
         say 'RESV ID     : ' substr(r, 13, 8)
         say 'GUEST NAME  : ' strip(substr(r, 21, 20))
         say 'ROOM NO     : ' substr(r, 41, 4)
         say 'ROOM TYPE   : ' substr(r, 45, 4)
         say 'CHECK-IN    : ' substr(r, 49, 4) || '-' || ,
                              substr(r, 53, 2) || '-' || substr(r, 55, 2)
         say 'CHECK-OUT   : ' substr(r, 57, 4) || '-' || ,
                              substr(r, 61, 2) || '-' || substr(r, 63, 2)
         say 'GUESTS      : ' substr(r, 65, 2)
         say 'NIGHTLY RATE: INR ' || (substr(r, 67, 6) + 0)
         say 'STATUS      : ' substr(r, 73, 1)
         say '=================================================='
         leave
      end
   end

   if found = 0 then do
      say 'HRBSUTIL: RESERVATION "'find_id'" NOT FOUND IN ' full_dsn
      exit 4
   end

#!/usr/bin/env python3
"""
Hotel Reservation and Billing System (HRBS) - Automated Source & Dataset Checker
Verifies:
  1. COBOL standards: 7-72 columns, Area A (cols 8-11), Area B (cols 12-72), comments in col 7.
  2. JCL standards: line lengths <= 71 characters, valid JOB card, matching IF/ELSE/ENDIF.
  3. Data standards: fixed-block record length consistency, non-empty keys, valid formatting.
  4. REXX standards: /* REXX */ comment header, balanced DO/END structures.
"""

import os
import sys
import glob

BASE_DIR = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
COBOL_DIR = os.path.join(BASE_DIR, "cobol")
JCL_DIR = os.path.join(BASE_DIR, "jcl")
DATA_DIR = os.path.join(BASE_DIR, "data")
REXX_DIR = os.path.join(BASE_DIR, "rexx")

def check_cobol_files():
    print("\n--- [1/4] CHECKING COBOL SOURCE CODE STANDARDS ---")
    cobol_files = glob.glob(os.path.join(COBOL_DIR, "*.cbl")) + glob.glob(os.path.join(COBOL_DIR, "dclgen", "*.cbl"))
    total_files = len(cobol_files)
    errors = []

    for fpath in cobol_files:
        rel_path = os.path.relpath(fpath, BASE_DIR)
        with open(fpath, "r", encoding="utf-8") as f:
            lines = f.readlines()

        for l_num, line in enumerate(lines, start=1):
            raw = line.rstrip("\r\n")
            if not raw:
                continue

            # Check max column 72 for code
            if len(raw) > 72:
                # If characters beyond 72 are non-space
                if len(raw[:72].rstrip()) != len(raw.rstrip()):
                    errors.append(f"{rel_path}:{l_num}: Line extends beyond column 72 ({len(raw)} chars): '{raw[:75]}...'")

            # Check Indicator Column (Column 7 -> index 6)
            if len(raw) >= 7:
                ind = raw[6]
                if ind not in [' ', '*', '/', '-', 'D', 'd']:
                    errors.append(f"{rel_path}:{l_num}: Invalid column 7 indicator '{ind}'")

    if errors:
        for e in errors:
            print(f"  [ERROR] {e}")
        return False
    else:
        print(f"  [PASS] All {total_files} COBOL programs strictly comply with columns 7-72 and Area A/B rules.")
        return True

def check_jcl_files():
    print("\n--- [2/4] CHECKING JCL STANDARDS AND SYNTAX ---")
    jcl_files = glob.glob(os.path.join(JCL_DIR, "*.jcl"))
    total_files = len(jcl_files)
    errors = []

    for fpath in jcl_files:
        rel_path = os.path.relpath(fpath, BASE_DIR)
        with open(fpath, "r", encoding="utf-8") as f:
            lines = f.readlines()

        has_job_card = False
        if_stack = []
        is_instream_data = False

        for l_num, line in enumerate(lines, start=1):
            raw = line.rstrip("\r\n")

            # Check instream data boundary
            if raw == "/*":
                is_instream_data = False
                continue

            if not is_instream_data and (" DD *" in raw or " DD DATA" in raw):
                is_instream_data = True

            # Standard JCL line length <= 71 (col 72 reserved for continuation/blank)
            if not is_instream_data and len(raw) > 71 and not raw.startswith("//*"):
                errors.append(f"{rel_path}:{l_num}: JCL line exceeds column 71 ({len(raw)} chars): '{raw[:75]}'")

            if raw.startswith("//") and " JOB " in raw:
                has_job_card = True

            # Match IF / THEN / ELSE / ENDIF
            if not is_instream_data and raw.startswith("//"):
                stripped = raw[2:].strip()
                if " IF " in stripped and " THEN" in stripped:
                    if_stack.append(l_num)
                elif stripped.startswith("IF ") and " THEN" in stripped:
                    if_stack.append(l_num)
                elif stripped.startswith("ENDIF") or " ENDIF" in stripped:
                    if if_stack:
                        if_stack.pop()
                    else:
                        errors.append(f"{rel_path}:{l_num}: Unmatched //ENDIF statement")

        if not has_job_card and "ALLOCATE" not in rel_path:
            errors.append(f"{rel_path}: Missing valid //JOB card")

        if if_stack:
            errors.append(f"{rel_path}: Unclosed //IF statement starting at line {if_stack[-1]}")

    if errors:
        for e in errors:
            print(f"  [ERROR] {e}")
        return False
    else:
        print(f"  [PASS] All {total_files} JCL jobs passed line length (<= 71), JOB card, and IF/ENDIF checks.")
        return True

def check_data_files():
    print("\n--- [3/4] CHECKING DATASET RECORD LENGTHS AND KEYS ---")
    data_files = [f for f in glob.glob(os.path.join(DATA_DIR, "*.dat")) if "EXPECTED" not in os.path.basename(f)]
    total_files = len(data_files)
    errors = []

    LRECL_MAP = {
        "RESERV.dat": 80,
        "FILE1.dat": 80,
        "FILE2.dat": 80,
        "FILE3.dat": 160,
        "INVOICE.dat": 100,

        "TRANS.dat": 80,
        "SERVICE.dat": 80,
        "RESERV_BAD.dat": 80,
        "SINGLE_REC.dat": 80,
        "MAX_LEN_NAMES.dat": 80,
        "PAY_MISMATCH.dat": 80,
        "EMPTY.dat": 80
    }

    for fpath in data_files:
        fname = os.path.basename(fpath)
        expected_lrecl = LRECL_MAP.get(fname, 80)
        rel_path = os.path.relpath(fpath, BASE_DIR)

        with open(fpath, "r", encoding="utf-8") as f:
            lines = [l.rstrip("\r\n") for l in f]

        if fname == "EMPTY.dat":
            if len(lines) != 0:
                errors.append(f"{rel_path}: EMPTY.dat should be empty but has {len(lines)} lines")
            continue

        for l_num, line in enumerate(lines, start=1):
            if len(line) != expected_lrecl:
                errors.append(f"{rel_path}:{l_num}: Line length {len(line)} != expected LRECL {expected_lrecl}")

    if errors:
        for e in errors:
            print(f"  [ERROR] {e}")
        return False
    else:
        print(f"  [PASS] All {total_files} datasets adhere strictly to fixed-block LRECL (80/100 bytes).")
        return True

def check_rexx_files():
    print("\n--- [4/4] CHECKING REXX SCRIPT STANDARDS ---")
    rexx_files = glob.glob(os.path.join(REXX_DIR, "*.rexx"))
    total_files = len(rexx_files)
    errors = []

    for fpath in rexx_files:
        rel_path = os.path.relpath(fpath, BASE_DIR)
        with open(fpath, "r", encoding="utf-8") as f:
            lines = f.readlines()

        if not lines or "/* REXX */" not in lines[0].upper():
            errors.append(f"{rel_path}: Missing mandatory /* REXX */ comment on line 1")

    if errors:
        for e in errors:
            print(f"  [ERROR] {e}")
        return False
    else:
        print(f"  [PASS] All {total_files} REXX procedures comply with standard TSO/E syntax.")
        return True

def main():
    print("=" * 70)
    print("HRBS CODEBASE AND DATASET INTEGRITY AUDITOR")
    print("=" * 70)

    c_ok = check_cobol_files()
    j_ok = check_jcl_files()
    d_ok = check_data_files()
    r_ok = check_rexx_files()

    print("\n" + "=" * 70)
    if c_ok and j_ok and d_ok and r_ok:
        print("ALL SOURCE FILES AND DATASETS PASSED STATIC VERIFICATION!")
        print("=" * 70)
        return 0
    else:
        print("VERIFICATION FAILED: ERRORS DETECTED IN CODEBASE.")
        print("=" * 70)
        return 1

if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env python3
"""
HRBS data loader generator.

Generates LOADDATA.jcl for the six datasets that can safely
be loaded through JCL in-stream data.

INVOICE.dat is deliberately excluded because its records are
100 bytes long and must be transferred separately.
"""

import os
import sys

BASE_DIR = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
DATA_DIR = os.path.join(BASE_DIR, "data")
JCL_DIR = os.path.join(BASE_DIR, "jcl")

# INVOICE.dat is intentionally excluded.
DATASETS = [
    ("RESERV.dat", "&SYSUID..HRBS.RESERV.INPUT", 80, 5),
    ("FILE1.dat", "&SYSUID..HRBS.FILE1", 80, 5),
    ("FILE2.dat", "&SYSUID..HRBS.FILE2", 80, 5),
    ("TRANS.dat", "&SYSUID..HRBS.TRANS.INPUT", 80, 5),
    ("SERVICE.dat", "&SYSUID..HRBS.SERVICE.INPUT", 80, 5),
    ("RESERV_BAD.dat", "&SYSUID..HRBS.RESERV.BAD", 80, 2),
]


def generate_load_jcl():
    os.makedirs(JCL_DIR, exist_ok=True)

    lines = [
        "//HRBSLOAD JOB (ACCT),'HRBS LOAD DATA',CLASS=A,MSGCLASS=X,",
        "//             NOTIFY=&SYSUID",
        "//* HRBS DATA LOADER GENERATED BY tools/make_load_jobs.py",
        "//* INVOICE DATA MUST BE TRANSFERRED SEPARATELY",
    ]

    for idx, (fname, dsn, lrecl, trk) in enumerate(DATASETS, start=1):
        fpath = os.path.join(DATA_DIR, fname)

        if not os.path.isfile(fpath):
            raise FileNotFoundError(f"Required input file is missing: {fpath}")

        with open(fpath, "r", encoding="utf-8", newline="") as f:
            records = [
                line.rstrip("\r\n")
                for line in f
                if line.strip()
            ]

        for rec_num, record in enumerate(records, start=1):
            if len(record) > lrecl:
                raise ValueError(
                    f"{fname}, record {rec_num}: length {len(record)} "
                    f"exceeds LRECL {lrecl}"
                )

            if record.startswith("//") or record.startswith("/*"):
                raise ValueError(
                    f"{fname}, record {rec_num}: begins with a JCL control "
                    f"sequence: {record[:10]!r}"
                )

            if "\t" in record:
                raise ValueError(f"{fname}, record {rec_num}: contains a tab")

        # Pad fixed-block records to their required logical record length.
        records = [record.ljust(lrecl) for record in records]

        lines.extend([
            f"//* STEP {idx:02d}: LOAD {fname} ({len(records)} RECORDS)",
            f"//LOAD{idx:02d} EXEC PGM=IEBGENER",
            "//SYSPRINT DD SYSOUT=*",
            "//SYSIN    DD DUMMY",
            f"//SYSUT2   DD DSN={dsn},DISP=(NEW,CATLG,DELETE),",
            f"//             SPACE=(TRK,({trk},{trk}),RLSE),",
            f"//             DCB=(RECFM=FB,LRECL={lrecl},BLKSIZE=0)",
            f"//SYSUT1   DD *,DCB=(RECFM=FB,LRECL={lrecl},BLKSIZE=0)",
        ])

        lines.extend(records)
        lines.append("/*")

    out_file = os.path.join(JCL_DIR, "LOADDATA.jcl")

    with open(out_file, "w", encoding="utf-8", newline="\n") as f:
        f.write("\n".join(lines) + "\n")

    print(
        f"Generated {out_file}: {len(DATASETS)} datasets checked. "
        "INVOICE.dat is excluded."
    )


if __name__ == "__main__":
    try:
        generate_load_jcl()
    except (OSError, ValueError) as exc:
        print(f"ERROR: {exc}", file=sys.stderr)
        sys.exit(1)

# 6. JOB08 & JOB09: VSAM KSDS Define, Initial Load & Transaction Update
# ----------------------------------------------------------------------
def run_job08_09(validated_reservations):
    print_jes_header("HRBSJ08", "IDCAMS VSAM KSDS Define & REPRO Initial Load")
    print("//IDCAMS01 EXEC PGM=IDCAMS")
    print("   [IDCAMS] DELETE &SYSUID..HRBS.RESERV.KSDS CLUSTER PURGE -> CC 0000")
    print("   [IDCAMS] DEFINE CLUSTER (KEYS(8 12), RECORDSIZE(80 80)) -> CC 0000")
    print(f"   [IDCAMS] REPRO INDATASET(RESERV.VALID) -> {len(validated_reservations)} RECORDS LOADED -> CC 0000")

    print_jes_header("HRBSJ09", "COBOL HRBSUPD: VSAM Random Transaction Engine")
    trans_file = os.path.join(DATA_DIR, "TRANS.dat")
    with open(trans_file, "r", encoding="utf-8") as f:
        trans_records = [line.rstrip("\r\n") for line in f if line.strip()]

    # Master KSDS in-memory state
    master = {r[12:20]: r for r in validated_reservations}
    active_bookings = []
    for r in validated_reservations:
        active_bookings.append({
            "room": r[40:44],
            "cin": r[48:56],
            "cout": r[56:64],
            "status": r[72:73]
        })

    adds = 0
    cancels = 0
    checkouts = 0
    updates = 0
    rejects = 0

    audit_log = []
    reject_log = []

    for t in trans_records:
        action = t[3:4]
        rid = t[4:12]
        name = t[12:32]
        room = t[32:36]
        rtype = t[36:40]
        cin = t[40:48]
        cout = t[48:56]
        gst = t[56:58]
        rate = t[58:64]

        if action == "A":
            if rid in master:
                rejects += 1
                reject_log.append(f"{t[:80]:<80}22DUPLICATE KEY - RESERVATION EXISTS    ")
                audit_log.append(f"20261005 REJ    {rid} {room} FAIL DUPLICATE KEY - RESERVATION EXISTS")
            else:
                # Check overlap
                overlap = False
                for b in active_bookings:
                    if b["room"] == room and b["status"] != "X":
                        if cin < b["cout"] and cout > b["cin"]:
                            overlap = True
                            break
                if overlap:
                    rejects += 1
                    reject_log.append(f"{t[:80]:<80}ROROOM NOT AVAILABLE - OVERLAPPING DATES ")
                    audit_log.append(f"20261005 REJ    {rid} {room} FAIL ROOM NOT AVAILABLE - OVERLAPPING DATES")
                else:
                    adds += 1
                    new_rec = f"RSVPUN261005{rid}{name}{room}{rtype}{cin}{cout}{gst}{rate}C       "
                    master[rid] = new_rec
                    active_bookings.append({"room": room, "cin": cin, "cout": cout, "status": "C"})
                    audit_log.append(f"20261005 ADD    {rid} {room} OK   RESERVATION ADDED SUCCESSFULLY")

        elif action == "C":
            if rid not in master:
                rejects += 1
                reject_log.append(f"{t[:80]:<80}23RESERVATION NOT FOUND FOR CANCELLATION")
                audit_log.append(f"20261005 REJ    {rid} {room} FAIL RESERVATION NOT FOUND FOR CANCELLATION")
            else:
                cancels += 1
                old = master[rid]
                master[rid] = old[:72] + "X" + old[73:]
                for b in active_bookings:
                    if b["room"] == old[40:44] and b["cin"] == old[48:56]:
                        b["status"] = "X"
                audit_log.append(f"20261005 CANCEL {rid} {room} OK   RESERVATION CANCELLED")

        elif action == "O":
            if rid not in master:
                rejects += 1
                reject_log.append(f"{t[:80]:<80}23RESERVATION NOT FOUND FOR CHECK-OUT   ")
                audit_log.append(f"20261005 REJ    {rid} {room} FAIL RESERVATION NOT FOUND FOR CHECK-OUT")
            else:
                checkouts += 1
                old = master[rid]
                master[rid] = old[:72] + "O" + old[73:]
                audit_log.append(f"20261005 CHKOUT {rid} {room} OK   GUEST CHECKED OUT")

        elif action == "U":
            if rid not in master:
                rejects += 1
                reject_log.append(f"{t[:80]:<80}23RESERVATION NOT FOUND FOR UPDATE      ")
                audit_log.append(f"20261005 REJ    {rid} {room} FAIL RESERVATION NOT FOUND FOR UPDATE")
            else:
                updates += 1
                old = master[rid]
                master[rid] = old[:48] + cin + cout + gst + rate + old[72:]
                audit_log.append(f"20261005 UPDATE {rid} {room} OK   RESERVATION DETAILS UPDATED")
        else:
            rejects += 1
            reject_log.append(f"{t[:80]:<80}99INVALID TRANSACTION ACTION CODE       ")
            audit_log.append(f"20261005 REJ    {rid} {room} FAIL INVALID TRANSACTION ACTION CODE")

    print("\n   +-------------------------------------------+")
    print("   | HRBSUPD - VSAM TRANSACTION UPDATE SUMMARY |")
    print(f"   | TRANSACTIONS READ   : {len(trans_records):05d}                 |")
    print(f"   | ADDS PROCESSED      : {adds:05d}                 |")
    print(f"   | CANCELS PROCESSED   : {cancels:05d}                 |")
    print(f"   | CHECK-OUTS PROCESSED: {checkouts:05d}                 |")
    print(f"   | UPDATES PROCESSED   : {updates:05d}                 |")
    print(f"   | REJECTED TRANS      : {rejects:05d}                 |")
    print("   +-------------------------------------------+")
    print(f"   [STATUS] JOB HRBSJ09 RETURN CODE: CC 0004 (DELIBERATE REJECTS AUDITED)")

# 7. JOB06: COBOL HRBSBIL Billing Engine
# ----------------------------------------------------------------------
def run_job06(validated_reservations):
    print_jes_header("HRBSJ06", "COBOL HRBSBIL: Billing Engine (Status C/O + Services)")
    srv_file = os.path.join(DATA_DIR, "SERVICE.dat")
    extra_srv = {}
    if os.path.exists(srv_file):
        with open(srv_file, "r", encoding="utf-8") as f:
            for line in f:
                if len(line.strip()) >= 22:
                    rid = line[0:8]
                    amt = int(line[14:22]) / 100.0
                    extra_srv[rid] = extra_srv.get(rid, 0.0) + amt

    invoices = []
    read_cnt = 0
    bill_cnt = 0
    skip_cnt = 0
    grand_total = 0.0
    seq = 0
    today = datetime.now().strftime("%Y%m%d")

    for r in validated_reservations:
        read_cnt += 1
        hotel = r[3:6]
        rid = r[12:20]
        name = r[20:40]
        cin = datetime.strptime(r[48:56], "%Y%m%d")
        cout = datetime.strptime(r[56:64], "%Y%m%d")
        rate = int(r[66:72])
        status = r[72:73]

        if status in ["C", "O"]:
            nights = max(1, (cout - cin).days)
            room = nights * rate
            base_svc = round(room * 0.05, 2)
            extra = extra_srv.get(rid, 0.0)
            svc = base_svc + extra
            tax_rate = 0.18 if rate > 7500 else 0.05
            tax = round((room + svc) * tax_rate, 2)
            tot = round(room + svc + tax, 2)
            seq += 1
            bill_cnt += 1
            grand_total += tot

            inv_rec = (
                "INV" + hotel + today +
                f"G{seq:09d}" +
                f"IN{seq:06d}" +
                rid + name +
                f"{nights:03d}" +
                f"{int(round(room * 100)):08d}" +
                f"{int(round(svc * 100)):08d}" +
                f"{int(round(tax * 100)):08d}" +
                f"{int(round(tot * 100)):010d}" +
                "P" + "  "
            )
            invoices.append(inv_rec)
        else:
            skip_cnt += 1

    out_file = os.path.join(DATA_DIR, "INVOICE.GENERATED.dat")
    with open(out_file, "w", newline="\n", encoding="utf-8") as f:
        f.write("\n".join(invoices) + "\n")

    print("\n   +-------------------------------------------+")
    print("   | HRBSBIL - BILLING RUN SUMMARY             |")
    print(f"   | RESERVATIONS READ : {read_cnt:05d}                 |")
    print(f"   | INVOICES WRITTEN  : {bill_cnt:05d}                 |")
    print(f"   | SKIPPED (NOT CONF): {skip_cnt:05d}                 |")
    print(f"   | EXTRA SRV LOADED  : {len(extra_srv):05d}                 |")
    print(f"   | TOTAL BILLED      : {grand_total:>14,.2f} INR   |")
    print("   +-------------------------------------------+")
    print(f"   [STATUS] JOB HRBSJ06 RETURN CODE: CC 0000")
    return invoices

# 9. JOB12: COBOL HRBSDB2 DB2 Loader & Cursor Report
# ----------------------------------------------------------------------
def run_job12(invoices, reservations):
    print_jes_header("HRBSJ12", "COBOL HRBSDB2: DB2 Embedded SQL Batch Load & Cursor Query")
    con = sqlite3.connect(":memory:")
    cur = con.cursor()

    schema_raw = open(os.path.join(DB2_DIR, "SCHEMA.sql"), encoding="utf-8").read()
    clean_lines = []
    for line in schema_raw.splitlines():
        if line.strip().startswith("CREATE DATABASE") or line.strip().startswith("CREATE TABLESPACE"):
            continue
        line = line.replace("IN HRBSDB.HRBSTS", "")
        line = line.replace("WITH DEFAULT", "DEFAULT")
        line = line.replace("HRBS.", "")
        clean_lines.append(line)
    cur.executescript("\n".join(clean_lines))

    # Insert reservations
    resv_ins = 0
    resv_dup = 0
    for r in reservations:
        try:
            rid = r[12:20]
            gid = "G000000001"
            rm = r[40:44]
            cin = f"{r[48:52]}-{r[52:54]}-{r[54:56]}"
            cout = f"{r[56:60]}-{r[60:62]}-{r[62:64]}"
            gst = int(r[64:66])
            stat = r[72:73]
            cur.execute("INSERT INTO RESERVATION VALUES (?, ?, ?, ?, ?, ?, ?)",
                        (rid, gid, rm, cin, cout, gst, stat))
            resv_ins += 1
        except sqlite3.IntegrityError:
            resv_dup += 1

    # Insert bills
    bill_ins = 0
    bill_dup = 0
    for inv in invoices:
        try:
            invid = inv[24:32]
            rid = inv[32:40]
            bdate = f"{inv[6:10]}-{inv[10:12]}-{inv[12:14]}"
            nts = int(inv[60:63])
            rmc = int(inv[63:71]) / 100.0
            svc = int(inv[71:79]) / 100.0
            tax = int(inv[79:87]) / 100.0
            tot = int(inv[87:97]) / 100.0
            cur.execute("INSERT INTO BILL VALUES (?, ?, ?, ?, ?, ?, ?, ?, 'P')",
                        (invid, rid, bdate, nts, rmc, svc, tax, tot))
            bill_ins += 1
        except sqlite3.IntegrityError:
            bill_dup += 1

    con.commit()

    print("\n   +-------------------------------------------+")
    print("   | HRBSDB2 - DB2 BATCH EXECUTION SUMMARY     |")
    print(f"   | RESERVATIONS INSERTED : {resv_ins:05d}             |")
    print(f"   | RESERVATIONS DUPLICATE: {resv_dup:05d}             |")
    print(f"   | BILLS INSERTED        : {bill_ins:05d}             |")
    print(f"   | BILLS DUPLICATE       : {bill_dup:05d}             |")
    print("   +-------------------------------------------+")

    print("\n   [CURSOR C_UNPAID] Querying Unpaid Invoices:")
    q = """
    SELECT B.INVOICE_ID, G.GUEST_NAME, B.TOTAL
      FROM BILL B, RESERVATION V, GUEST G
     WHERE B.RES_ID = V.RES_ID AND V.GUEST_ID = G.GUEST_ID
       AND B.TOTAL > (SELECT COALESCE(SUM(P.AMOUNT), 0)
                        FROM PAYMENT P WHERE P.RES_ID = B.RES_ID)
     LIMIT 5;
    """
    for row in cur.execute(q):
        print(f"   -> INVOICE: {row[0]} | GUEST: {row[1]:<20} | OUTSTANDING: INR {row[2]:>10,.2f}")
    print(f"   [STATUS] JOB HRBSJ12 RETURN CODE: CC 0000")
